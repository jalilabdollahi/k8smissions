Install in order, then fix and restart the operator:

kubectl -n k8smissions get cm release-manifests -o jsonpath='{.data.<KEY>}' | kubectl apply -f -
kubectl wait --for=condition=Established crd/workloads.platform.acme.io
kubectl -n k8smissions edit configmap workload-operator-code     # a missing optional field is None, not an error
kubectl -n k8smissions rollout restart deploy/workload-operator
