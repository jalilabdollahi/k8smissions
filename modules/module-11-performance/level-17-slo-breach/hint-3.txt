    {"op":"replace","path":"/spec/template/spec/containers/0/livenessProbe","value":{"httpGet":{"path":"/live","port":8080},"periodSeconds":<about 5>,"timeoutSeconds":1,"failureThreshold":<2 or 3>}}]'
